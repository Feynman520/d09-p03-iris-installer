# IRIS 설치 패키지 구현 계획 v1 (2026-09-10)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> 서브에이전트에게 넘길 때 함께 전달할 루트 규칙: ① Anthropic API 프록시 우회 절대 금지(`ANTHROPIC_BASE_URL` 유지) ② 프로세스 종료는 PID로만(이름 기반 일괄 킬 금지) ③ 한글·백슬래시가 든 파일은 Bash heredoc이 아니라 Write/Edit 도구로 쓴다(R-002) ④ 파이썬 인라인 실행은 `PYTHONUTF8=1`(R-010).

**Goal:** 아무것도 설치되지 않은 Windows PC에서 `IRIS-설치.zip` 하나를 풀고 `IRIS-설치.cmd`를 두 번 클릭하면, 구독 로그인 한 번을 빼고 IRIS 체계 전체(동봉 런타임·에이전트·TeamClaude·IRIS-Face·세팅가이드 v10)가 끝까지 설치되고 Face가 첫 세팅 세션을 열어 준다.

**Architecture:** 겉옷(`IRIS-설치.cmd` → `bootstrap.ps1`, ASCII 전용)이 무결성 검사 후 동봉 Node를 풀어 속옷(`installer/server.mjs`, 127.0.0.1:3460 로컬 웹 화면)을 띄운다. 속옷이 이름·구독·복사·로그인을 처리하고 영수증(`package-receipt.json`)과 최소 표지(`soul-state.json`)를 남긴 뒤 IRIS-Face를 `--first-session`으로 열어 에이전트에게 세팅가이드 v10을 넘긴다. zip은 이 PC의 공장(`build/`)이 `lock.json`대로 부품을 모아 정화·지문·포장하고, 검사소(`verify/`)가 정적·재현성·실행 3층으로 검증한다.

**Tech Stack:** Node 24 ESM(설치기·공장·검사소·Face), `node:test`(단위 시험, 외부 의존 0), PowerShell 5.1(겉옷만), Windows 내장 `tar.exe`(zip 해제), IRIS-Face(P02, Electron 44·node-pty), TeamClaude 1.1.16 + 로컬 패치 4파일, 세팅가이드 v10(`_setup-guides/_guide-tools/Build-V8Guide.py`로 재생성).

**Spec:** `<P03 리포 루트>\docs\설계.md` (v1, 2026-09-10 사용자 확정)

```js
import { createHash } from 'node:crypto'; import fs from 'node:fs'; import path from 'node:path';
export function sha256File(file) {
  return new Promise((res, rej) => { const h = createHash('sha256'); fs.createReadStream(file).on('data', d => h.update(d)).on('end', () => res(h.digest('hex'))).on('error', rej); });
}
export async function buildManifest({ payloadDir, lock, faceVersion }) {
  const parts = {};
  for (const [name, p] of Object.entries(lock.parts)) {
    const files = p.file.endsWith('/') ? fs.readdirSync(path.join(payloadDir, p.file)).map(f => p.file + f) : [p.file];
    for (const rel of files) {
      const abs = path.join(payloadDir, rel);
      parts[files.length > 1 ? `${name}:${path.basename(rel)}` : name] = { file: rel, version: name === 'face' ? faceVersion : (p.version ?? null), sha256: await sha256File(abs), bytes: fs.statSync(abs).size };
    }
  }
  return { schema: 1, built: new Date().toISOString(), package: { name: 'IRIS', version: lock.package.version, guideVersion: lock.package.guideVersion, license: 'MIT' }, parts };
}
export async function verifyManifest(payloadDir, manifest) {
  const mismatches = [];
  for (const p of Object.values(manifest.parts)) {
    const abs = path.join(payloadDir, p.file);
    const actual = fs.existsSync(abs) ? await sha256File(abs) : null;
    if (actual !== p.sha256) mismatches.push({ file: p.file, expected: p.sha256, actual });
  }
  return { ok: mismatches.length === 0, mismatches };
}
```

```js
import { spawn } from 'node:child_process';
export function run(exe, args, { cwd, env = process.env, timeoutMs = 600000, stdin = 'ignore' } = {}) {
  return new Promise((resolve) => {
    const c = spawn(exe, args, { cwd, env, windowsHide: true, stdio: [stdin, 'pipe', 'pipe'] });
    let out = '', err = ''; c.stdout.on('data', d => out += d); c.stderr.on('data', d => err += d);
    const t = setTimeout(() => c.kill(), timeoutMs);
    c.on('close', code => { clearTimeout(t); resolve({ code, out: out.trim(), err: err.trim() }); });
    c.on('error', e => { clearTimeout(t); resolve({ code: -1, out: '', err: e.message }); });
  });
}
```

`build/collect.mjs` 핵심:
```js
export async function collect({ lock, cacheDir, stageDir, nodeDir, log }) {
  const payloadDir = path.join(stageDir, 'payload'); fs.rmSync(stageDir, { recursive: true, force: true }); fs.mkdirSync(payloadDir, { recursive: true });
  const skipped = []; let faceVersion = null;
  for (const [name, p] of Object.entries(lock.parts)) {
    const dest = path.join(payloadDir, p.file); fs.mkdirSync(path.dirname(dest), { recursive: true });
    if (p.kind === 'url') {
      if (!p.sha256) throw new Error(`sha256 missing for ${name}`);
      const cached = path.join(cacheDir, path.basename(p.file));
      if (!fs.existsSync(cached) || await sha256File(cached) !== p.sha256) { log(`download ${name}`); await download(p.url, cached); }
      if (await sha256File(cached) !== p.sha256) throw new Error(`sha256 mismatch for ${name}`);
      fs.copyFileSync(cached, dest);
    } else if (p.kind === 'npm-prefix') {
      if (p.redistribute === 'download') { skipped.push(name); continue; }
      const prefix = path.join(stageDir, 'npm', name); fs.mkdirSync(prefix, { recursive: true });
      const r = await run(process.execPath, [npmCli(nodeDir), 'install', '-g', '--prefix', prefix, `${p.npm}@${p.version}`, '--no-fund', '--no-audit'], { env: { ...process.env, npm_config_cache: path.join(cacheDir, 'npm-cache') } });
      if (r.code !== 0) throw new Error(`npm install ${name}: ${r.err}`);
      if (p.patches) await applyPatches(prefix, JSON.parse(fs.readFileSync(p.patches, 'utf8')), log);   // Task 6
      await zipDir(prefix, dest);
    } else if (p.kind === 'dir') {
      const work = path.join(stageDir, 'dir', name); copyTree(p.source, work, p.exclude ?? []);
      if (p.npmCi) { const r = await run(process.execPath, [npmCli(nodeDir), 'ci', '--no-fund', '--no-audit'], { cwd: work }); if (r.code !== 0) throw new Error(`npm ci ${name}: ${r.err}`); }
      if (name === 'face') faceVersion = JSON.parse(fs.readFileSync(path.join(work, 'package.json'), 'utf8')).version;
      await zipDir(work, dest);
    } else if (p.kind === 'file') fs.copyFileSync(p.source, dest);
    else if (p.kind === 'glob') { for (const f of fs.readdirSync(p.source).filter(f => globMatch(f, p.pattern))) fs.copyFileSync(path.join(p.source, f), path.join(payloadDir, p.file, f)); }
  }
  return { payloadDir, faceVersion, skipped };
}
```
`download`는 `fetch` + `fs.createWriteStream`, `copyTree`는 `fs.cpSync(src, dst, { recursive: true, filter })`, `npmCli(nodeDir)`은 `path.join(nodeDir, 'node_modules/npm/bin/npm-cli.js')`(nodeDir 미지정이면 이 PC의 `npm root -g`의 상위 `npm-cli.js`), `globMatch`는 `*`만 지원하는 정규식 변환.

```js
export async function applyPatches(prefixDir, rules, log) {
  for (const f of rules.files) {
```

```powershell
param([Parameter(Mandatory)][string]$ZipRoot, [int]$Port = 3460)
$ErrorActionPreference = 'Stop'
$work = Join-Path $env:LOCALAPPDATA 'IRIS-Installer'; New-Item -ItemType Directory -Force $work | Out-Null
$log = Join-Path $work 'bootstrap.log'; function Log($m) { "$(Get-Date -Format s) $m" | Add-Content $log; Write-Host $m }
# 10: running from inside a zip (Explorer temp) or a path that lacks payload
if ($ZipRoot -like "$env:TEMP*" -or -not (Test-Path (Join-Path $ZipRoot 'payload\manifest.json'))) { Log 'Please extract the zip first, then run IRIS-설치.cmd from the extracted folder.'; exit 10 }
# 11: integrity (manifest sha256 vs files) - PowerShell 5.1 Get-FileHash
$mf = Get-Content (Join-Path $ZipRoot 'payload\manifest.json') -Raw | ConvertFrom-Json
foreach ($p in $mf.parts.PSObject.Properties) { $f = Join-Path $ZipRoot ('payload\' + $p.Value.file)
  if (-not (Test-Path $f) -or (Get-FileHash $f -Algorithm SHA256).Hash.ToLower() -ne $p.Value.sha256) { Log "Integrity check failed: $($p.Value.file). Download the zip again."; exit 11 } }
# 12: portable node
$nodeZip = Join-Path $ZipRoot ('payload\' + $mf.parts.node.file); $nodeDir = Join-Path $work 'node'
if (-not (Test-Path (Join-Path $nodeDir 'node.exe'))) { Remove-Item -Recurse -Force $nodeDir -ErrorAction SilentlyContinue; New-Item -ItemType Directory -Force $nodeDir | Out-Null
  & "$env:SystemRoot\System32\tar.exe" -xf $nodeZip -C $nodeDir --strip-components 1; if ($LASTEXITCODE -ne 0) { Log 'Could not unpack Node.'; exit 12 } }
# 13: server
$server = Join-Path $ZipRoot 'installer\server.mjs'
$p = Start-Process -FilePath (Join-Path $nodeDir 'node.exe') -ArgumentList @("`"$server`"", '--zip-root', "`"$ZipRoot`"", '--port', $Port, '--node-dir', "`"$nodeDir`"") -WindowStyle Hidden -PassThru -RedirectStandardOutput (Join-Path $work 'server.out.log') -RedirectStandardError (Join-Path $work 'server.err.log')
"$($p.Id)" | Set-Content (Join-Path $work 'server.pid')
$ok = $false; for ($i = 0; $i -lt 40 -and -not $ok; $i++) { Start-Sleep -Milliseconds 500; try { $r = Invoke-WebRequest "http://127.0.0.1:$Port/api/health" -UseBasicParsing -TimeoutSec 2; $ok = $r.StatusCode -eq 200 } catch {} }
if (-not $ok) { Log 'Setup server did not respond.'; exit 13 }
Start-Process "http://127.0.0.1:$Port/"; Log "Opened browser (server pid $($p.Id))"; exit 0
```
`--strip-components 1`은 `node-v24.17.0-win-x64\` 한 겹을 벗긴다. Node가 이미 풀려 있으면 재사용(재실행 안전).
